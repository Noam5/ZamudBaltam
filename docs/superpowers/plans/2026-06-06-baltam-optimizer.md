# Baltam Optimizer Tab — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a read-only "ייעול בלת״ם" tab to `index.html` that interactively shows how much per-trip Baltam can be eliminated via smart dispatch, per-doctor appointment flexibility, and a part-time second car.

**Architecture:** Two canonical CommonJS-UMD modules — `optdata.js` (embedded 312-trip dataset + price table + hours, generated by `gen_optdata.js`) and `optimizer.js` (pure scheduling math) — are unit-tested in Node, then injected verbatim into `index.html` between markers by `build_opt.js` (so the embedded code can never drift from the tested code). The tab's UI (`renderOpt`) lives in `index.html` and calls the injected globals.

**Tech Stack:** Vanilla JS (no framework), `xlsx` already vendored, Node for tests/build. No new dependencies.

---

## File structure

- `gen_optdata.js` (create) — reads `live_state.json` + `classified.json`, normalizes doctors, adds 4 routes, writes `optdata.js`. Run once / when source data changes.
- `optdata.js` (generated) — `OPT_TRIPS`, `OPT_TRAVEL`, `OPT_HOURS`, `OPT_DOCTORS`. UMD.
- `optimizer.js` (create) — pure functions: helpers, `greedyBaltam`, `optimizeAll`, `recommendWindows`. UMD.
- `optimizer.test.js` (create) — Node assertions against validated numbers.
- `build_opt.js` (create) — injects `optimizer.js` + `optdata.js` source into `index.html` between `<!-- OPT-INJECT:START -->` / `<!-- OPT-INJECT:END -->`.
- `index.html` (modify) — tab button, `#tab-opt` content, inject markers, `renderOpt()` + tab-switch hook.

```js
// optimizer.js — pure Baltam-optimization functions (UMD: Node require + browser globals).
(function (root, factory) {
  const api = factory();
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else Object.assign(root, api);
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  function timeToMin(t) { if (!t) return 0; const p = t.split(':'); return (+p[0]) * 60 + (+p[1]); }
  function endMinOf(t) { const r = timeToMin(t.endTime), s = timeToMin(t.startTime); return (r === 0 && s > 0) ? 1440 : r; }
  function dowOf(d) { return new Date(d + 'T00:00:00').getDay(); }

  function makeLookups(travel) {
    function travelTime(from, to, morning) {
      for (const e of travel) if ((e.from === from && e.to === to) || (e.from === to && e.to === from)) return morning ? e.morningMin : e.afternoonMin;
      return null;
    }
    function tripPrice(from, to) {
      for (const e of travel) if ((e.from === from && e.to === to) || (e.from === to && e.to === from)) return e.check ? 0 : (e.price || 0);
      return 0;
    }
    return { travelTime, tripPrice };
  }

  // Exact port of index.html classifyTrips → summed B/BML/B120 price (the app's current Baltam).
  function greedyBaltam(trips, hours, travel) {
    const { travelTime, tripPrice } = makeLookups(travel);
    const sorted = trips.map((t, i) => Object.assign({ _i: i }, t))
      .sort((a, b) => a.date !== b.date ? a.date.localeCompare(b.date) : a.startTime.localeCompare(b.startTime));
    const res = []; let cum = 0, total = 0; const byMonth = {};
    for (let i = 0; i < sorted.length; i++) {
      const t = sorted[i], s = timeToMin(t.startTime), rE = timeToMin(t.endTime);
      const eDur = (rE === 0 && s > 0) ? 1440 : rE, dur = eDur - s, dw = dowOf(t.date);
      const wh = dw === 6 ? null : hours[dw];
      let code = 'Z';
      if (wh) { const ws = timeToMin(wh.start), we = timeToMin(wh.end); if (s < ws || rE > we) code = 'BML'; } else code = 'BML';
      if (code !== 'BML') {
        let pz = null; for (let j = i - 1; j >= 0; j--) if (res[j].date === t.date && res[j].code === 'Z') { pz = res[j]; break; }
        if (pz) {
          const pr = timeToMin(pz.endTime), pe = (pr === 0 && timeToMin(pz.startTime) > 0) ? 1440 : pr;
          const tm = travelTime(pz.to, t.from, s < 720);
          if (tm !== null) code = (pe - s + tm > 0) ? 'B' : 'Z';
        }
      }
      if (code === 'Z') { cum += dur; if (cum > 210 * 60) code = 'B120'; }
      const price = code === 'Z' ? 0 : tripPrice(t.from, t.to);
      res.push(Object.assign({ code, price }, t));
      if (code !== 'Z') { total += price; const m = t.date.slice(0, 7); byMonth[m] = (byMonth[m] || 0) + price; }
    }
    return { total, byMonth };
  }

  return { timeToMin, endMinOf, dowOf, makeLookups, greedyBaltam };
});
```

```js
  // Per-day dispatch: primary car (within working hours) + optional secondary car
  // (only inside its windows), each trip's pickup shiftable within ±tol. Returns
  // { baltam, plan:[{trip,car,start}] }. Earliest-feasible start; full-state memo.
  function optimizeDay(dayTrips, ctx) {
    const { travelTime, tripPrice } = ctx.lk, hours = ctx.hours;
    const tolOf = t => (ctx.tolByDoctor && ctx.tolByDoctor[t.doctor] != null) ? ctx.tolByDoctor[t.doctor] : (ctx.defaultTol || 0);
    const dw = dowOf(dayTrips[0].date), wh = dw === 6 ? null : hours[dw];
    const ws = wh ? timeToMin(wh.start) : 0, we = wh ? timeToMin(wh.end) : 0;
    const wins = (ctx.secondaryWindows || []).filter(w => w.dow === dw);
    const ts = dayTrips.map(t => { const s = timeToMin(t.startTime), e = endMinOf(t); return { t, s, e, dur: e - s, tl: tolOf(t), price: tripPrice(t.from, t.to), morning: s < 720 }; })
      .sort((a, b) => (a.s - a.tl) - (b.s - b.tl) || a.e - b.e);
    const n = ts.length;
    // earliest start for cand on a car whose last trip ended at (prevTime,prevLoc), within [availStart,availEnd]
    function place(cand, prevTime, prevLoc, availStart, availEnd) {
      let lo = cand.s - cand.tl;
      if (prevLoc != null) { const tm = travelTime(prevLoc, cand.t.from, cand.morning); if (tm === null) return null; lo = Math.max(lo, prevTime + tm); }
      let st = Math.max(lo, cand.s - cand.tl, availStart);
      if (st > cand.s + cand.tl) return null;
      if (st + cand.dur > availEnd) return null;
      return st;
    }
    const memo = new Map();
    function rec(i, t1, l1, t2, l2) {
      if (i === n) return { val: 0, plan: [] };
      const key = i + '|' + t1 + '|' + l1 + '|' + t2 + '|' + l2;
      const hit = memo.get(key); if (hit) return hit;
      const c = ts[i];
      let best = rec(i + 1, t1, l1, t2, l2); best = { val: best.val, plan: best.plan }; // skip
      if (wh) {
        const st = place(c, t1, l1 == null ? null : l1, ws, we);
        if (st != null) { const sub = rec(i + 1, st + c.dur, c.t.to, t2, l2), v = c.price + sub.val; if (v > best.val) best = { val: v, plan: [{ trip: c.t, car: 1, start: st }].concat(sub.plan) }; }
      }
      for (const w of wins) {
        const st = place(c, t2, l2 == null ? null : l2, w.start, w.end);
        if (st != null) { const sub = rec(i + 1, t1, l1, st + c.dur, c.t.to), v = c.price + sub.val; if (v > best.val) best = { val: v, plan: [{ trip: c.t, car: 2, start: st }].concat(sub.plan) }; }
      }
      memo.set(key, best); return best;
    }
    const r = rec(0, -1, null, -1, null);
    const totalPrice = ts.reduce((a, b) => a + b.price, 0);
    return { baltam: totalPrice - r.val, plan: r.plan };
  }

  function optimizeAll(opts) {
    const ctx = { lk: makeLookups(opts.travel), hours: opts.hours, tolByDoctor: opts.tolByDoctor, defaultTol: opts.defaultTol, secondaryWindows: opts.secondaryWindows };
    const byDate = {}; for (const t of opts.trips) (byDate[t.date] = byDate[t.date] || []).push(t);
    let baltam = 0; const plan = [];
    for (const d of Object.keys(byDate).sort()) { const r = optimizeDay(byDate[d], ctx); baltam += r.baltam; for (const p of r.plan) plan.push(p); }
    return { baltam, plan };
  }
```

Note: in `place`, `prevTime` of `-1` with `prevLoc` of `null` means "car unused"; since `prevLoc==null` the travel branch is skipped, so the `-1` is never used arithmetically.

```js
  // Run with an unlimited 2nd car, then bucket the trips it serves into (dow,band) windows.
  function recommendWindows(opts) {
    const sec = [0, 1, 2, 3, 4, 5, 6].map(dw => ({ dow: dw, start: 0, end: 1440 }));
    const r = optimizeAll(Object.assign({}, opts, { secondaryWindows: sec }));
    const { tripPrice } = makeLookups(opts.travel);
    const band = s => s < 11 * 60 ? 'AM' : s < 14 * 60 ? 'MID' : 'PM';
    const groups = {};
    for (const p of r.plan) {
      if (p.car !== 2) continue;
      const dw = dowOf(p.trip.date), b = band(p.start), key = dw + '|' + b;
      const e = endMinOf(p.trip), price = tripPrice(p.trip.from, p.trip.to);
      const g = groups[key] || (groups[key] = { dow: dw, band: b, start: 1e9, end: 0, count: 0, price: 0 });
      g.start = Math.min(g.start, p.start); g.end = Math.max(g.end, p.start + (e - timeToMin(p.trip.startTime)));
      g.count++; g.price += price;
    }
    const windows = Object.values(groups).sort((a, b) => b.price - a.price);
    return { windows, totalPrice: windows.reduce((a, w) => a + w.price, 0) };
  }
```

```js
    function optToggleWin(i) { optState.windows[i].enabled = !optState.windows[i].enabled; renderOpt(); }
    function optEditWin(i, field, v) { const m = (v || '').match(/^(\d{1,2}):(\d{2})$/); if (m) optState.windows[i][field] = (+m[1]) * 60 + (+m[2]); renderOpt(); }
    function optToggleSmart() { optState.smartDispatch = !optState.smartDispatch; renderOpt(); }
    const mm = x => String(Math.floor(x / 60)).padStart(2, '0') + ':' + String(x % 60).padStart(2, '0');
    const HEBDOW = ['א', 'ב', 'ג', 'ד', 'ה', 'ו', 'ש'];

    function renderOpt() {
      if (!optState) optState = optDefaultState();
      const enabledWins = optState.windows.filter(w => w.enabled).map(w => ({ dow: w.dow, start: w.start, end: w.end }));
      const baseGreedy = greedyBaltam(OPT_TRIPS, OPT_HOURS, OPT_TRAVEL).total;
      // staged attribution: dispatch -> flexibility -> 2nd car
      const afterDispatch = optState.smartDispatch ? optimizeAll({ trips: OPT_TRIPS, hours: OPT_HOURS, travel: OPT_TRAVEL, defaultTol: 0 }).baltam : baseGreedy;
      const afterFlex = optState.smartDispatch
        ? optimizeAll({ trips: OPT_TRIPS, hours: OPT_HOURS, travel: OPT_TRAVEL, tolByDoctor: optState.tol }).baltam
        : afterDispatch;
      const finalRun = optimizeAll({ trips: OPT_TRIPS, hours: OPT_HOURS, travel: OPT_TRAVEL, tolByDoctor: optState.smartDispatch ? optState.tol : {}, defaultTol: optState.smartDispatch ? 0 : 0, secondaryWindows: enabledWins });
      const finalBaltam = optState.smartDispatch ? finalRun.baltam : afterDispatch; // if smart off, ignore levers
      const saved = baseGreedy - finalBaltam;
      const dispSave = baseGreedy - afterDispatch, flexSave = afterDispatch - afterFlex, carSave = afterFlex - finalBaltam;
      const winHrsWk = optState.windows.filter(w => w.enabled).reduce((a, w) => a + (w.end - w.start) / 60, 0); // per representative week
      const carMo = carSave / 4;

      // ---- controls ----
      let c = '<div class="alias-panel-title">ידיות</div>';
      c += `<label style="display:block;margin:6px 0"><input type="checkbox" ${optState.smartDispatch ? 'checked' : ''} onchange="optToggleSmart()"> שיבוץ חכם של הרכב (חינם)</label>`;
      c += '<div style="margin:8px 0 4px;font-weight:600">גמישות פגישות (± דקות)</div>';
      c += '<div style="margin-bottom:4px">' + [['none', 'ללא'], ['cons', 'שמרני'], ['real', 'מציאותי'], ['aggr', 'אגרסיבי']].map(([k, l]) => `<button class="btn" style="color:#333;border-color:#999;font-size:12px;padding:2px 8px" onclick="optPreset('${k}')">${l}</button>`).join(' ') + '</div>';
      c += '<table class="alias-table"><tbody>';
      for (const d of OPT_DOCTORS) c += `<tr><td>${esc(d)}</td><td><select onchange="optSetTol('${esc(d).replace(/'/g, "\\'")}',this.value)">${[0, 15, 30, 60].map(v => `<option value="${v}" ${optState.tol[d] === v ? 'selected' : ''}>±${v}</option>`).join('')}</select></td></tr>`;
      c += '</tbody></table>';
      c += '<div style="margin:8px 0 4px;font-weight:600">חלונות רכב משני</div>';
      c += '<table class="alias-table"><thead><tr><th></th><th>יום</th><th>משעה</th><th>עד</th><th>₪/חודש</th></tr></thead><tbody>';
      optState.windows.forEach((w, i) => {
        c += `<tr><td><input type="checkbox" ${w.enabled ? 'checked' : ''} onchange="optToggleWin(${i})"></td><td>${HEBDOW[w.dow]}</td>` +
          `<td><input type="text" value="${mm(w.start)}" style="width:48px" onchange="optEditWin(${i},'start',this.value)"></td>` +
          `<td><input type="text" value="${mm(w.end)}" style="width:48px" onchange="optEditWin(${i},'end',this.value)"></td>` +
          `<td>${Math.round((w._price || 0) / 4)}</td></tr>`;
      });
      c += '</tbody></table>';
      document.getElementById('optControls').innerHTML = c;

      // ---- results ----
      let r = '<div class="summary-bar">' +
        `<div class="summary-item"><span class="summary-label">בלת״ם נוכחי:</span><span class="summary-value baltam">${baseGreedy} ₪ (${Math.round(baseGreedy / 4)}/חודש)</span></div>` +
        `<div class="summary-item"><span class="summary-label">לאחר ייעול:</span><span class="summary-value zamud">${finalBaltam} ₪ (${Math.round(finalBaltam / 4)}/חודש)</span></div>` +
        `<div class="summary-item"><span class="summary-label">חיסכון:</span><span class="summary-value zamud">${saved} ₪ (${Math.round(saved / 4)}/חודש · ${Math.round(saved / baseGreedy * 100)}%)</span></div>` +
        '</div>';
      r += '<div class="legend" style="display:block"><b>תרומת כל ידית (4 חודשים):</b> ' +
        `שיבוץ חכם ₪${dispSave} · גמישות ₪${flexSave} · רכב משני ₪${carSave}</div>`;
      if (winHrsWk > 0) r += `<div class="settings-note">רכב משני: ${winHrsWk.toFixed(1)} שעות/שבוע, חוסך ₪${Math.round(carMo)}/חודש → כדאי אם רכב חלקי בחלונות אלו עולה פחות מ-₪${Math.round(carMo)} לחודש (≈ ₪${winHrsWk ? Math.round(carMo / (winHrsWk * 4.33)) : 0}/שעה).</div>`;
      r += '<div class="settings-note">מספרי גמישות הם <b>אומדן</b>.</div>';

      // rescued trips table
      const rescued = optState.smartDispatch ? finalRun.plan : [];
      r += '<div class="table-wrap" style="max-height:calc(100vh - 260px)"><table><thead><tr><th>#</th><th>תאריך</th><th>יום</th><th>ממקום</th><th>למקום</th><th>רופא</th><th>רכב</th><th>שעה חדשה</th></tr></thead><tbody>';
      rescued.forEach((p, i) => {
        r += `<tr><td>${i + 1}</td><td>${toDisplayDate(p.trip.date)}</td><td>${HEBDOW[getDayOfWeek(p.trip.date)]}</td><td>${esc(p.trip.from)}</td><td>${esc(p.trip.to)}</td><td>${esc(p.trip.doctor)}</td><td>${p.car === 2 ? 'משני' : 'צמוד'}</td><td>${mm(p.start)}</td></tr>`;
      });
      r += '</tbody></table></div>';
      document.getElementById('optResults').innerHTML = r;
    }
```
